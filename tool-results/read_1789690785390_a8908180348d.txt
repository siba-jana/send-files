    81→async function copyText(text: string, successMessage: string): Promise<void> {
    82→  try {
    83→    if (navigator.clipboard?.writeText) {
    84→      await navigator.clipboard.writeText(text);
    85→    } else {
    86→      throw new Error("Clipboard API unavailable");
    87→    }
    88→    toast.success(successMessage);
    89→  } catch {
    90→    try {
    91→      const textarea = document.createElement("textarea");
    92→      textarea.value = text;
    93→      textarea.style.position = "fixed";
    94→      textarea.style.opacity = "0";
    95→      document.body.appendChild(textarea);
    96→      textarea.select();
    97→      document.execCommand("copy");
    98→      textarea.remove();
    99→      toast.success(successMessage);
   100→    } catch {
   101→      toast.error("Couldn't copy — please copy it manually.");
   102→    }
   103→  }
   104→}

   126→function formatRemaining(ms: number): string {
   127→  if (ms <= 0) return "Expired";
   128→  const totalSeconds = Math.floor(ms / 1000);
   129→  const hours = Math.floor(totalSeconds / 3600);
   130→  const minutes = Math.floor((totalSeconds % 3600) / 60);
   131→  const seconds = totalSeconds % 60;
   132→  if (hours > 0) return `${hours} h ${minutes.toString().padStart(2, "0")} min`;
   133→  if (minutes > 0) return `${minutes} min ${seconds.toString().padStart(2, "0")} s`;
   134→  return `${seconds} s`;
   135→}

   137→/** Live "Expires in 23 h 59 min" label. Starts null to avoid SSR mismatch. */
   138→function useCountdown(expiresAt: string | null | undefined): string | null {
   139→  const [snapshot, setSnapshot] = useState<{
   140→    expiresAt: string;
   141→    label: string;
   142→  } | null>(null);
   143→
   144→  useEffect(() => {
   145→    if (!expiresAt) return;
   146→    const target = new Date(expiresAt).getTime();
   147→    const tick = () => {
   148→      setSnapshot({ expiresAt, label: formatRemaining(target - Date.now()) });
   149→    };
   150→    tick();
   151→    const id = window.setInterval(tick, 1000);
   152→    return () => window.clearInterval(id);
   153→  }, [expiresAt]);
   154→
   155→  return snapshot && snapshot.expiresAt === expiresAt
   156→    ? snapshot.label
   157→    : null;
   158→}

   594→            ref={attachFolderInput}
   595→            id="send-folder-input"
   596→            type="file"
   597→            multiple
   598→            aria-label="Choose a folder to send"
   599→            className="sr-only"
   600→            onChange={handleInputChange}
   601→          />
   602→        </div>
   603→
   604→        {/* Inline errors while composing */}
   605→        {error && editing && (
   606→          <Alert variant="destructive" className="mt-4">
   607→            <TriangleAlert />
   608→            <AlertDescription>{error}</AlertDescription>
   609→          </Alert>
   610→        )}
   611→
   612→        {/* Large transfer warning */}
   613→        {files.length > 0 && totalBytes > LARGE_TRANSFER_BYTES && (
   614→          <Alert className="mt-4 border-amber-300 bg-amber-50 text-amber-900 dark:border-amber-500/30 dark:bg-amber-500/10 dark:text-amber-200">
   615→            <TriangleAlert aria-hidden="true" />
   616→            <AlertDescription className="text-amber-900 dark:text-amber-200">
   617→              Large transfer — keep both browsers open and awake.
   618→            </AlertDescription>
   619→          </Alert>
   620→        )}
   621→
   622→        {/* File list */}
   623→        {files.length > 0 && (
   624→          <div className="mt-6">
   625→            <ul
   626→              aria-label="Files to send"
   627→              className="max-h-72 divide-y overflow-y-auto pr-1 [scrollbar-width:thin] [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-border [&::-webkit-scrollbar-track]:transparent"
   628→            >
   629→              {files.map(({ key, file }) => (
   630→                <li
   631→                  key={key}
   632→                  className="flex items-center gap-3 py-2.5 first:pt-0"
   633→                >
   634→                  <FileIcon name={file.name} mimeType={file.type} />
   635→                  <div className="min-w-0 flex-1">
   636→                    <p className="truncate text-sm font-medium" title={file.name}>
   637→                      {file.name}
   638→                    </p>
   639→                    <p className="text-xs text-muted-foreground tabular-nums">
   640→                      {formatBytes(file.size)}
   641→                    </p>
   642→                  </div>
   643→                  <Button
   644→                    type="button"
   645→                    variant="ghost"
   646→                    size="icon"
   647→                    className="size-9 shrink-0 text-muted-foreground hover:text-destructive"
   648→                    aria-label={`Remove ${file.name}`}
   649→                    disabled={creating}
   650→                    onClick={() => removeFile(key)}
   651→                  >
   652→                    <X aria-hidden="true" />
   653→                  </Button>
   654→                </li>
   655→              ))}
   656→            </ul>
   657→            <div className="mt-1 flex items-center justify-between border-t pt-3">
   658→              <p className="text-sm text-muted-foreground tabular-nums">
   659→                {files.length} file{files.length === 1 ? "" : "s"} •{" "}
   660→                {formatBytes(totalBytes)}
   661→              </p>
   662→              <Button
   663→                type="button"
   664→                variant="ghost"
   665→                size="sm"
   666→                className="h-9 text-muted-foreground hover:text-destructive"
   667→                disabled={creating}
   668→                onClick={clearFiles}
   669→              >
   670→                <X aria-hidden="true" />
   671→                Clear
   672→              </Button>
   673→            </div>
   674→          </div>
   675→        )}
   676→
   677→        <Separator className="my-5" />
   678→
   679→        {/* Transfer options */}
   680→        <Collapsible open={optionsOpen} onOpenChange={setOptionsOpen}>
   681→          <CollapsibleTrigger asChild>
   682→            <Button
   683→              type="button"
   684→              variant="ghost"
   685→              className="h-11 w-full justify-between px-2 text-muted-foreground hover:text-foreground"
   686→            >
   687→              <span className="inline-flex items-center gap-2">
   688→                <Settings2 aria-hidden="true" className="size-4" />
   689→                Transfer options
   690→              </span>
   691→              <svg
   692→                aria-hidden="true"
   693→                viewBox="0 0 24 24"
   694→                fill="none"
   695→                stroke="currentColor"
   696→                strokeWidth="2"
   697→                strokeLinecap="round"
   698→                strokeLinejoin="round"
   699→                className={cn(
   700→                  "size-4 transition-transform duration-200",
   701→                  optionsOpen && "rotate-180",
   702→                )}
   703→              >
   704→                <path d="m6 9 6 6 6-6" />
   705→              </svg>
   706→            </Button>
   707→          </CollapsibleTrigger>
   708→          <CollapsibleContent className="space-y-4 pt-1">
   709→            <div className="grid gap-4 sm:grid-cols-2">
   710→              <div className="space-y-2.5 sm:col-span-2">
   711→                <div className="flex items-center justify-between gap-3 rounded-xl border p-3.5">
   712→                  <div className="flex items-start gap-2.5">
   713→                    <Lock
   714→                      aria-hidden="true"
   715→                      className="mt-0.5 size-4 shrink-0 text-muted-foreground"
   716→                    />
   717→                    <div>
   718→                      <Label
   719→                        htmlFor="use-password-switch"
   720→                        className="text-sm font-medium"
   721→                      >
   722→                        Password protect
   723→                      </Label>
   724→                      <p className="mt-0.5 text-xs text-muted-foreground">
   725→                        Recipients must enter the password to unlock the files.
   726→                      </p>
   727→                    </div>
   728→                  </div>
   729→                  <Switch
   730→                    id="use-password-switch"
   731→                    checked={options.usePassword}
   732→                    onCheckedChange={(checked) =>
   733→                      setOption("usePassword", checked)
   734→                    }
   735→                  />
   736→                </div>
   737→                {options.usePassword && (
   738→                  <div className="space-y-1.5">
   739→                    <Label htmlFor="transfer-password" className="text-sm font-medium">
   740→                      Password
   741→                    </Label>
   742→                    <Input
   743→                      id="transfer-password"
   744→                      type="password"
   745→                      value={options.password}
   746→                      autoComplete="new-password"
   747→                      placeholder="At least 4 characters"
   748→                      className="h-11 rounded-xl"
   749→                      aria-invalid={passwordTooShort || undefined}
   750→                      onChange={(e) => setOption("password", e.target.value)}
   751→                    />
   752→                    <p
   753→                      className={cn(
   754→                        "text-xs",
   755→                        passwordTooShort
   756→                          ? "text-destructive"
   757→                          : "text-muted-foreground",
   758→                      )}
   759→                    >
   760→                      {passwordTooShort
   761→                        ? "Password must be at least 4 characters."
   762→                        : "Minimum 4 characters."}
   763→                    </p>
   764→                  </div>
   765→                )}
   766→              </div>
   767→
   768→              <div className="space-y-2">
   769→                <Label htmlFor="expiry-select" className="text-sm font-medium">
   770→                  Link expires after
   771→                </Label>
   772→                <Select
   773→                  value={String(options.expiresInHours)}
   774→                  onValueChange={(value) =>
   775→                    setOption("expiresInHours", Number(value))
   776→                  }
   777→                >
   778→                  <SelectTrigger
   779→                    id="expiry-select"
   780→                    className="h-11 w-full rounded-xl"
   781→                  >
   782→                    <SelectValue />
   783→                  </SelectTrigger>
   784→                  <SelectContent>
   785→                    {EXPIRY_OPTIONS.map((option) => (
   786→                      <SelectItem key={option.value} value={option.value}>
   787→                        {option.label}
   788→                      </SelectItem>
   789→                    ))}
   790→                  </SelectContent>
   791→                </Select>
   792→              </div>
   793→
   794→              <div className="space-y-2">
   795→                <Label
   796→                  htmlFor="max-downloads-select"
   797→                  className="text-sm font-medium"
   798→                >
   799→                  Max downloads
   800→                </Label>
   801→                <Select
   802→                  value={String(options.maxDownloads)}
   803→                  onValueChange={(value) =>
   804→                    setOption("maxDownloads", Number(value))
   805→                  }
   806→                >
   807→                  <SelectTrigger
   808→                    id="max-downloads-select"
   809→                    className="h-11 w-full rounded-xl"
   810→                  >
   811→                    <SelectValue />
   812→                  </SelectTrigger>
   813→                  <SelectContent>
   814→                    {Array.from({ length: 10 }, (_, i) => i + 1).map((n) => (
   815→                      <SelectItem key={n} value={String(n)}>
   816→                        {n}
   817→                      </SelectItem>
   818→                    ))}
   819→                  </SelectContent>
   820→                </Select>
   821→              </div>
   822→
   823→              <div className="space-y-2 sm:col-span-2">
   824→                <Label htmlFor="sender-name" className="text-sm font-medium">
   825→                  Your name{" "}
   826→                  <span className="font-normal text-muted-foreground">
   827→                    (optional)
   828→                  </span>
   829→                </Label>
   830→                <Input
   831→                  id="sender-name"
   832→                  value={options.senderName}
   833→                  maxLength={40}
   834→                  placeholder="Anonymous sender"
   835→                  className="h-11 rounded-xl"
   836→                  onChange={(e) => setOption("senderName", e.target.value)}
   837→                />
   838→              </div>
   839→            </div>
   840→          </CollapsibleContent>
   841→        </Collapsible>
   842→
   843→        {/* Create */}
   844→        <Button
   845→          type="button"
   846→          size="lg"
   847→          disabled={files.length === 0 || creating}
   848→          onClick={() => void create()}
   849→          className="mt-4 h-12 w-full rounded-xl bg-rose-600 text-base text-white shadow-lg shadow-rose-600/25 hover:bg-rose-700 dark:bg-rose-500 dark:shadow-rose-500/20 dark:hover:bg-rose-600"
   850→        >
   851→          {creating ? (
   852→            <>
   853→              <LoaderCircle aria-hidden="true" className="animate-spin" />
   854→              Creating transfer…
   855→            </>
   856→          ) : (
   857→            <>
   858→              <Upload aria-hidden="true" />
   859→              Create Transfer
   860→            </>
   861→          )}
   862→        </Button>
   863→
   864→        <p className="mt-4 text-center text-xs leading-relaxed text-muted-foreground">
   865→          Files transfer directly between connected browsers — nothing is
   866→          uploaded until a recipient connects.
   867→        </p>
   868→      </CardContent>
   869→    </Card>
   870→  );
