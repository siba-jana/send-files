    81→async function copyText(text: string, successMessage: string): Promise<void> {
    82→  try {
    83→    if (navigator.clipboard?.writeText) {
    84→      await navigator.clipboard.writeText(text);
    85→    } else {
    86→      throw new Error("Clipboard API unavailable");
    87→    }
    88→    toast.success(successMessage);
    89→  } catch {
    90→    try {
    91→      const textarea = document.createElement("textarea");
    92→      textarea.value = text;
    93→      textarea.style.position = "fixed";
    94→      textarea.style.opacity = "0";
    95→      document.body.appendChild(textarea);
    96→      textarea.select();
    97→      document.execCommand("copy");
    98→      textarea.remove();
    99→      toast.success(successMessage);
   100→    } catch {
   101→      toast.error("Couldn't copy — please copy it manually.");
   102→    }
   103→  }
   104→}

   106→/** Live "Expires in 23 h 59 min" label. Starts null to avoid SSR mismatch. */
   107→function useCountdown(expiresAt: string | null | undefined): string | null {
   108→  const [label, setLabel] = useState<string | null>(null);
   109→
   110→  useEffect(() => {
   111→    if (!expiresAt) {
   112→      setLabel(null);
   113→      return;
   114→    }
   115→    const target = new Date(expiresAt).getTime();
   116→    const tick = () => {
   117→      const remaining = target - Date.now();
   118→      if (remaining <= 0) {
   119→        setLabel("Expired");
   120→        return;
   121→      }
   122→      const totalSeconds = Math.floor(remaining / 1000);
   123→      const hours = Math.floor(totalSeconds / 3600);
   124→      const minutes = Math.floor((totalSeconds % 3600) / 60);
   125→      const seconds = totalSeconds % 60;
   126→      setLabel(
   127→        hours > 0
   128→          ? `${hours} h ${minutes.toString().padStart(2, "0")} min`
   129→          : minutes > 0
   130→            ? `${minutes} min ${seconds.toString().padStart(2, "0")} s`
   131→            : `${seconds} s`,
   132→      );
   133→    };
   134→    tick();
   135→    const id = window.setInterval(tick, 1000);
   136→    return () => window.clearInterval(id);
   137→  }, [expiresAt]);
   138→
   139→  return label;
   140→}

   596→            <TriangleAlert />
   597→            <AlertDescription>{error}</AlertDescription>
   598→          </Alert>
   599→        )}
   600→
   601→        {/* Large transfer warning */}
   602→        {files.length > 0 && totalBytes > LARGE_TRANSFER_BYTES && (
   603→          <Alert className="mt-4 border-amber-300 bg-amber-50 text-amber-900 dark:border-amber-500/30 dark:bg-amber-500/10 dark:text-amber-200">
   604→            <TriangleAlert aria-hidden="true" />
   605→            <AlertDescription className="text-amber-900 dark:text-amber-200">
   606→              Large transfer — keep both browsers open and awake.
   607→            </AlertDescription>
   608→          </Alert>
   609→        )}
   610→
   611→        {/* File list */}
   612→        {files.length > 0 && (
   613→          <div className="mt-6">
   614→            <ul
   615→              aria-label="Files to send"
   616→              className="max-h-72 divide-y overflow-y-auto pr-1 [scrollbar-width:thin] [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-border [&::-webkit-scrollbar-track]:transparent"
   617→            >
   618→              {files.map(({ key, file }) => (
   619→                <li
   620→                  key={key}
   621→                  className="flex items-center gap-3 py-2.5 first:pt-0"
   622→                >
   623→                  <FileIcon name={file.name} mimeType={file.type} />
   624→                  <div className="min-w-0 flex-1">
   625→                    <p className="truncate text-sm font-medium" title={file.name}>
   626→                      {file.name}
   627→                    </p>
   628→                    <p className="text-xs text-muted-foreground tabular-nums">
   629→                      {formatBytes(file.size)}
   630→                    </p>
   631→                  </div>
   632→                  <Button
   633→                    type="button"
   634→                    variant="ghost"
   635→                    size="icon"
   636→                    className="size-9 shrink-0 text-muted-foreground hover:text-destructive"
   637→                    aria-label={`Remove ${file.name}`}
   638→                    disabled={!editing && !creating}
   639→                    onClick={() => removeFile(key)}
   640→                  >
   641→                    <X aria-hidden="true" />
   642→                  </Button>
   643→                </li>
   644→              ))}
   645→            </ul>
   646→            <div className="mt-1 flex items-center justify-between border-t pt-3">
   647→              <p className="text-sm text-muted-foreground tabular-nums">
   648→                {files.length} file{files.length === 1 ? "" : "s"} •{" "}
   649→                {formatBytes(totalBytes)}
   650→              </p>
   651→              <Button
   652→                type="button"
   653→                variant="ghost"
   654→                size="sm"
   655→                className="h-9 text-muted-foreground hover:text-destructive"
   656→                disabled={!editing && !creating}
   657→                onClick={clearFiles}
   658→              >
   659→                <X aria-hidden="true" />
   660→                Clear
   661→              </Button>
   662→            </div>
   663→          </div>
   664→        )}
   665→
   666→        <Separator className="my-5" />
   667→
   668→        {/* Transfer options */}
   669→        <Collapsible open={optionsOpen} onOpenChange={setOptionsOpen}>
   670→          <CollapsibleTrigger asChild>
   671→            <Button
   672→              type="button"
   673→              variant="ghost"
   674→              className="h-11 w-full justify-between px-2 text-muted-foreground hover:text-foreground"
   675→            >
   676→              <span className="inline-flex items-center gap-2">
   677→                <Settings2 aria-hidden="true" className="size-4" />
   678→                Transfer options
   679→              </span>
   680→              <svg
   681→                aria-hidden="true"
   682→                viewBox="0 0 24 24"
   683→                fill="none"
   684→                stroke="currentColor"
   685→                strokeWidth="2"
   686→                strokeLinecap="round"
   687→                strokeLinejoin="round"
   688→                className={cn(
   689→                  "size-4 transition-transform duration-200",
   690→                  optionsOpen && "rotate-180",
   691→                )}
   692→              >
   693→                <path d="m6 9 6 6 6-6" />
   694→              </svg>
   695→            </Button>
   696→          </CollapsibleTrigger>
   697→          <CollapsibleContent className="space-y-4 pt-1">
   698→            <div className="grid gap-4 sm:grid-cols-2">
   699→              <div className="space-y-2.5 sm:col-span-2">
   700→                <div className="flex items-center justify-between gap-3 rounded-xl border p-3.5">
   701→                  <div className="flex items-start gap-2.5">
   702→                    <Lock
   703→                      aria-hidden="true"
   704→                      className="mt-0.5 size-4 shrink-0 text-muted-foreground"
   705→                    />
   706→                    <div>
   707→                      <Label
   708→                        htmlFor="use-password-switch"
   709→                        className="text-sm font-medium"
   710→                      >
   711→                        Password protect
   712→                      </Label>
   713→                      <p className="mt-0.5 text-xs text-muted-foreground">
   714→                        Recipients must enter the password to unlock the files.
   715→                      </p>
   716→                    </div>
   717→                  </div>
   718→                  <Switch
   719→                    id="use-password-switch"
   720→                    checked={options.usePassword}
   721→                    onCheckedChange={(checked) =>
   722→                      setOption("usePassword", checked)
   723→                    }
   724→                  />
   725→                </div>
   726→                {options.usePassword && (
   727→                  <div className="space-y-1.5">
   728→                    <Label htmlFor="transfer-password" className="text-sm font-medium">
   729→                      Password
   730→                    </Label>
   731→                    <Input
   732→                      id="transfer-password"
   733→                      type="password"
   734→                      value={options.password}
   735→                      autoComplete="new-password"
   736→                      placeholder="At least 4 characters"
   737→                      className="h-11 rounded-xl"
   738→                      aria-invalid={passwordTooShort || undefined}
   739→                      onChange={(e) => setOption("password", e.target.value)}
   740→                    />
   741→                    <p
   742→                      className={cn(
   743→                        "text-xs",
   744→                        passwordTooShort
   745→                          ? "text-destructive"
   746→                          : "text-muted-foreground",
   747→                      )}
   748→                    >
   749→                      {passwordTooShort
   750→                        ? "Password must be at least 4 characters."
   751→                        : "Minimum 4 characters."}
   752→                    </p>
   753→                  </div>
   754→                )}
   755→              </div>
   756→
   757→              <div className="space-y-2">
   758→                <Label htmlFor="expiry-select" className="text-sm font-medium">
   759→                  Link expires after
   760→                </Label>
   761→                <Select
   762→                  value={String(options.expiresInHours)}
   763→                  onValueChange={(value) =>
   764→                    setOption("expiresInHours", Number(value))
   765→                  }
   766→                >
   767→                  <SelectTrigger
   768→                    id="expiry-select"
   769→                    className="h-11 w-full rounded-xl"
   770→                  >
   771→                    <SelectValue />
   772→                  </SelectTrigger>
   773→                  <SelectContent>
   774→                    {EXPIRY_OPTIONS.map((option) => (
   775→                      <SelectItem key={option.value} value={option.value}>
   776→                        {option.label}
   777→                      </SelectItem>
   778→                    ))}
   779→                  </SelectContent>
   780→                </Select>
   781→              </div>
   782→
   783→              <div className="space-y-2">
   784→                <Label
   785→                  htmlFor="max-downloads-select"
   786→                  className="text-sm font-medium"
   787→                >
   788→                  Max downloads
   789→                </Label>
   790→                <Select
   791→                  value={String(options.maxDownloads)}
   792→                  onValueChange={(value) =>
   793→                    setOption("maxDownloads", Number(value))
   794→                  }
   795→                >
   796→                  <SelectTrigger
   797→                    id="max-downloads-select"
   798→                    className="h-11 w-full rounded-xl"
   799→                  >
   800→                    <SelectValue />
   801→                  </SelectTrigger>
   802→                  <SelectContent>
   803→                    {Array.from({ length: 10 }, (_, i) => i + 1).map((n) => (
   804→                      <SelectItem key={n} value={String(n)}>
   805→                        {n}
   806→                      </SelectItem>
   807→                    ))}
   808→                  </SelectContent>
   809→                </Select>
   810→              </div>
   811→
   812→              <div className="space-y-2 sm:col-span-2">
   813→                <Label htmlFor="sender-name" className="text-sm font-medium">
   814→                  Your name{" "}
   815→                  <span className="font-normal text-muted-foreground">
   816→                    (optional)
   817→                  </span>
   818→                </Label>
   819→                <Input
   820→                  id="sender-name"
   821→                  value={options.senderName}
   822→                  maxLength={40}
   823→                  placeholder="Anonymous sender"
   824→                  className="h-11 rounded-xl"
   825→                  onChange={(e) => setOption("senderName", e.target.value)}
   826→                />
   827→              </div>
   828→            </div>
   829→          </CollapsibleContent>
   830→        </Collapsible>
   831→
   832→        {/* Create */}
   833→        <Button
   834→          type="button"
   835→          size="lg"
   836→          disabled={files.length === 0 || creating}
   837→          onClick={() => void create()}
   838→          className="mt-4 h-12 w-full rounded-xl bg-rose-600 text-base text-white shadow-lg shadow-rose-600/25 hover:bg-rose-700 dark:bg-rose-500 dark:shadow-rose-500/20 dark:hover:bg-rose-600"
   839→        >
   840→          {creating ? (
   841→            <>
   842→              <LoaderCircle aria-hidden="true" className="animate-spin" />
   843→              Creating transfer…
   844→            </>
   845→          ) : (
   846→            <>
   847→              <Upload aria-hidden="true" />
   848→              Create Transfer
   849→            </>
   850→          )}
   851→        </Button>
   852→
   853→        <p className="mt-4 text-center text-xs leading-relaxed text-muted-foreground">
   854→          Files transfer directly between connected browsers — nothing is
   855→          uploaded until a recipient connects.
   856→        </p>
   857→      </CardContent>
   858→    </Card>
   859→  );
   860→}
