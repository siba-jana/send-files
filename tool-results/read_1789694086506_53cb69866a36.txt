    76→function formatRemaining(ms: number): string {
    77→  if (ms <= 0) return "Expired";
    78→  const totalSeconds = Math.floor(ms / 1000);
    79→  const hours = Math.floor(totalSeconds / 3600);
    80→  const minutes = Math.floor((totalSeconds % 3600) / 60);
    81→  const seconds = totalSeconds % 60;
    82→  if (hours > 0) return `${hours} h ${minutes.toString().padStart(2, "0")} min`;
    83→  if (minutes > 0) return `${minutes} min ${seconds.toString().padStart(2, "0")} s`;
    84→  return `${seconds} s`;
    85→}

    87→/** Live "Expires in 23 h 59 min" label. Starts null to avoid SSR mismatch. */
    88→function useCountdown(expiresAt: string | null | undefined): string | null {
    89→  const [snapshot, setSnapshot] = useState<{
    90→    expiresAt: string;
    91→    label: string;
    92→  } | null>(null);
    93→
    94→  useEffect(() => {
    95→    if (!expiresAt) return;
    96→    const target = new Date(expiresAt).getTime();
    97→    const tick = () => {
    98→      setSnapshot({ expiresAt, label: formatRemaining(target - Date.now()) });
    99→    };
   100→    tick();
   101→    const id = window.setInterval(tick, 1000);
   102→    return () => window.clearInterval(id);
   103→  }, [expiresAt]);
   104→
   105→  return snapshot && snapshot.expiresAt === expiresAt
   106→    ? snapshot.label
   107→    : null;
   108→}

   590→          <div className="flex flex-col gap-2.5">
   591→            {zipSupported ? (
   592→              <Tooltip>

   594→                  <Button

   596→                    className="h-11 w-full rounded-xl bg-rose-600 text-white shadow-lg shadow-rose-600/25 hover:bg-rose-700 dark:bg-rose-500 dark:shadow-rose-500/20 dark:hover:bg-rose-600"

   619→                  "h-11 w-full rounded-xl",

   621→                    "bg-rose-600 text-white shadow-lg shadow-rose-600/25 hover:bg-rose-700 dark:bg-rose-500 dark:shadow-rose-500/20 dark:hover:bg-rose-600",

   623→                onClick={() => downloadAll(results)}

   625→                <Download aria-hidden="true" />
   626→                Download files separately{downloadable.length > 1 ? ` (${downloadable.length})` : ""}
   627→              </Button>
   628→            )}
   629→            <Button
   630→              type="button"
   631→              variant={zipSupported ? "outline" : "default"}
   632→              className={cn(
   633→                "h-11 w-full rounded-xl",
   634→                !zipSupported &&
   635→                  "bg-rose-600 text-white shadow-lg shadow-rose-600/25 hover:bg-rose-700 dark:bg-rose-500 dark:shadow-rose-500/20 dark:hover:bg-rose-600",

   640→              }}
   641→            >
   642→              <RotateCcw aria-hidden="true" />
   643→              Receive More Files
   644→            </Button>
   645→          </div>
   646→        </CardContent>
   647→      </Card>

   649→  }

   651→  // ----------------------------------------------------- cancelled | failed
   652→
   653→  if (phase === "cancelled") {

   672→            Enter Another Code

   678→
   679→  if (phase === "failed") {

   693→            onClick={reset}
   694→          >
   695→            <RotateCcw aria-hidden="true" />
   696→            Try Again
   697→          </Button>
   698→        </CardContent>

   702→
   703→  // ------------------------------------------------------------- idle
   704→
   705→  return (

   708→        <div className="text-center">

   730→            aria-label="Transfer code"
   731→            className="h-16 w-full rounded-2xl border-2 bg-background text-center text-3xl font-semibold tracking-[0.5em] ps-[0.25em] tabular-nums transition-colors outline-none placeholder:text-muted-foreground/50 focus-visible:border-rose-500 focus-visible:ring-4 focus-visible:ring-rose-500/20 dark:focus-visible:border-rose-400"
   732→          />
   733→          <Button
   734→            type="button"
   735→            size="lg"
   736→            disabled={normalizeCode(code).length !== 6}
   737→            onClick={submitCode}
   738→            className="h-12 w-full rounded-xl bg-rose-600 text-base text-white shadow-lg shadow-rose-600/25 hover:bg-rose-700 dark:bg-rose-500 dark:shadow-rose-500/20 dark:hover:bg-rose-600"
   739→          >
   740→            <Download aria-hidden="true" />
   741→            Receive Files
   742→          </Button>
   743→        </div>
   744→
   745→        <div className="my-6 flex items-center gap-3" aria-hidden="true">
   746→          <Separator className="flex-1" />
   747→          <span className="text-xs text-muted-foreground">or</span>
   748→          <Separator className="flex-1" />
   749→        </div>
   750→

   776→            </Button>

   780→        <p className="mt-5 text-center text-xs leading-relaxed text-muted-foreground">
